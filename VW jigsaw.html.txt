<script>
/* Responsive 3x5 Jigsaw with SVG clipPaths (classic tabs+blanks) + snap bounce
   - 3 rows x 5 cols
   - 2 attempts total
   - 60s per attempt
   - Slightly transparent dark-gray outlines
   - Responsive: pieces recompute on resize
   - Silent bounce snap when a correct piece is placed
*/

(function(){
  // config
  const ROWS = 3, COLS = 5, TOTAL = ROWS*COLS;
  const MAX_ATTEMPTS = 2, ATTEMPT_TIME = 60;

  // DOM
  const appRoot = document.getElementById('appRoot');
  const variantSel = document.getElementById('variantSel');
  const startBtn = document.getElementById('startBtn');
  const hintBtn = document.getElementById('hintBtn');
  const retryBtn = document.getElementById('retryBtn');
  const board = document.getElementById('board');
  const pool = document.getElementById('pool');
  const imgFile = document.getElementById('imgFile');
  const imgUrl = document.getElementById('imgUrl');
  const loadBtn = document.getElementById('loadBtn');
  const message = document.getElementById('message');
  const attemptsBadge = document.getElementById('attemptsBadge');
  const timerBadge = document.getElementById('timerBadge');

  // state
  let imgSrc = '';
  let attempt = 0;
  let timeLeft = ATTEMPT_TIME;
  let timerId = null;
  let mapping = []; // randomized order of pieces
  let edgeMap = []; // for each piece: {top: -1|0|1, right: -1|0|1, bottom, left}
  let placedMap = {}; // slotIndex -> pieceId
  let pieceElements = {}; // pieceId -> svgElement
  let solved = false;

  // helpers
  function show(msg){ message.textContent = msg; }
  function updateBadges(){ attemptsBadge.textContent = `Attempts: ${attempt} / ${MAX_ATTEMPTS}`; timerBadge.textContent = `Time: ${timeLeft}s`; }

  variantSel.addEventListener('change', ()=> applyVariant(variantSel.value));
  function applyVariant(v){ if(v==='minimal'){ appRoot.classList.add('minimal'); appRoot.classList.remove('withHeader'); } else { appRoot.classList.remove('minimal'); appRoot.classList.add('withHeader'); } }

  // build complementary edge map for the grid
  function generateEdgeMap(){
    edgeMap = new Array(TOTAL).fill(null).map(()=>({top:0,right:0,bottom:0,left:0}));
    for(let r=0;r<ROWS;r++){
      for(let c=0;c<COLS-1;c++){
        const idx = r*COLS + c;
        const rightTab = Math.random() > 0.5 ? 1 : -1;
        edgeMap[idx].right = rightTab;
        edgeMap[idx+1].left = -rightTab;
      }
    }
    for(let r=0;r<ROWS-1;r++){
      for(let c=0;c<COLS;c++){
        const idx = r*COLS + c;
        const bottomTab = Math.random() > 0.5 ? 1 : -1;
        edgeMap[idx].bottom = bottomTab;
        edgeMap[idx + COLS].top = -bottomTab;
      }
    }
  }

  function getBoardGeometry(){
    const rect = board.getBoundingClientRect();
    const boardW = rect.width, boardH = rect.height;
    const pieceW = boardW / COLS, pieceH = boardH / ROWS;
    return {boardW,boardH,pieceW,pieceH};
  }

  // SVG path generator for jigsaw piece
  function makePiecePath(w,h,edges, tabSizeFactor=0.22, tabDepthFactor=0.14){
    const ts = Math.min(w,h) * tabSizeFactor;
    const td = Math.min(w,h) * tabDepthFactor;
    function horizEdge(x0,x1,y,dir){ if(dir===0) return `L ${x1} ${y} `;
      const mid = (x0+x1)/2; const left = mid - ts/2, right = mid+ts/2; const depth = dir * td; return `L ${left} ${y} C ${left + ts*0.08} ${y} ${mid - ts*0.02} ${y + depth} ${mid} ${y + depth} C ${mid + ts*0.02} ${y + depth} ${right - ts*0.08} ${y} ${right} ${y} L ${x1} ${y} `; }
    function vertEdge(y0,y1,x,dir){ if(dir===0) return `L ${x} ${y1} `; const mid = (y0+y1)/2; const top = mid - ts/2, bottom = mid + ts/2; const depth = dir * td; return `L ${x} ${top} C ${x} ${top + ts*0.08} ${x + depth} ${mid - ts*0.02} ${x + depth} ${mid} C ${x + depth} ${mid + ts*0.02} ${x} ${bottom - ts*0.08} ${x} ${bottom} L ${x} ${y1} `; }
    function horizEdgeRightToLeft(x0,x1,y,dir){ if(dir===0) return `L ${x1} ${y} `; const mid = (x0+x1)/2; const left = mid - ts/2, right = mid + ts/2; const depth = -dir * td; return `L ${right} ${y} C ${right - ts*0.08} ${y} ${mid + ts*0.02} ${y + depth} ${mid} ${y + depth} C ${mid - ts*0.02} ${y + depth} ${left + ts*0.08} ${y} ${left} ${y} L ${x1} ${y} `; }
    function vertEdgeBottomToTop(y0,y1,x,dir){ if(dir===0) return `L ${x} ${y1} `; const mid = (y0+y1)/2; const top = mid - ts/2, bottom = mid + ts/2; const depth = -dir * td; return `L ${x} ${bottom} C ${x} ${bottom - ts*0.08} ${x + depth} ${mid + ts*0.02} ${x + depth} ${mid} C ${x + depth} ${mid - ts*0.02} ${x} ${top + ts*0.08} ${x} ${top} L ${x} ${y1} `; }
    let d = `M 0 0 `;
    d += horizEdge(0,w,0, edges.top);
    d += vertEdge(0,h,w, edges.right);
    d += horizEdgeRightToLeft(w,0,h, edges.bottom);
    d += vertEdgeBottomToTop(h,0,0, edges.left);
    d += 'Z';
    return d;
  }

  // build svg piece
  function buildPieceSVG(r,c,boardW,boardH,pieceW,pieceH,edges,pieceId){
    const svgns = "http://www.w3.org/2000/svg";
    const svg = document.createElementNS(svgns,'svg');
    svg.setAttribute('width', pieceW); svg.setAttribute('height', pieceH); svg.setAttribute('viewBox', `0 0 ${pieceW} ${pieceH}`);
    svg.classList.add('piece-svg'); svg.dataset.pieceId = pieceId; svg.style.width = pieceW + 'px'; svg.style.height = pieceH + 'px'; svg.style.position = 'absolute';
    const clipId = `clip-${pieceId}`;
    const defs = document.createElementNS(svgns,'defs');
    const clipPath = document.createElementNS(svgns,'clipPath'); clipPath.setAttribute('id', clipId);
    const path = document.createElementNS(svgns,'path'); const d = makePiecePath(pieceW, pieceH, edges);
    path.setAttribute('d', d); path.setAttribute('fill','white'); path.setAttribute('stroke','none'); clipPath.appendChild(path); defs.appendChild(clipPath); svg.appendChild(defs);
    const imgEl = document.createElementNS(svgns,'image'); const xOff = -c * pieceW; const yOff = -r * pieceH;
    imgEl.setAttributeNS(null,'href', imgSrc); imgEl.setAttribute('x', xOff); imgEl.setAttribute('y', yOff); imgEl.setAttribute('width', boardW); imgEl.setAttribute('height', boardH); imgEl.setAttribute('clip-path', `url(#${clipId})`);
    svg.appendChild(imgEl);
    const outline = document.createElementNS(svgns,'path'); outline.setAttribute('d', d); outline.setAttribute('fill','none'); outline.setAttribute('stroke', 'rgba(15,15,15,0.55)'); outline.setAttribute('stroke-width', Math.max(1, Math.min(pieceW, pieceH) * 0.02)); outline.setAttribute('vector-effect','non-scaling-stroke'); svg.appendChild(outline);
    return svg;
  }

  function createMapping(){ mapping = [...Array(TOTAL).keys()]; for(let i=mapping.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [mapping[i],mapping[j]]=[mapping[j],mapping[i]]; } }

  function buildBoardAndPieces(){
    board.innerHTML = ''; pool.innerHTML = ''; placedMap = {}; pieceElements = {};
    const geom = getBoardGeometry(); const {boardW, boardH, pieceW, pieceH} = geom;
    board.style.position = 'relative'; board.style.gridTemplateColumns = `repeat(${COLS}, 1fr)`; board.style.gridTemplateRows = `repeat(${ROWS}, 1fr)`;
    for(let r=0;r<ROWS;r++){ for(let c=0;c<COLS;c++){ const slot = document.createElement('div'); slot.className='slot'; slot.style.width='100%'; slot.style.height='100%'; slot.dataset.index = r*COLS + c; const left = c * pieceW; const top = r * pieceH; slot.dataset.left = left; slot.dataset.top = top; slot.addEventListener('dragover', e=>e.preventDefault()); slot.addEventListener('drop', onDropSlot); board.appendChild(slot); } }
    createMapping();
    for(let i=0;i<TOTAL;i++){
      const visualIndex = mapping[i]; const r = Math.floor(visualIndex / COLS); const c = visualIndex % COLS; const pieceId = `piece-${visualIndex}-${Date.now()}-${i}`;
      const svg = buildPieceSVG(r,c,boardW,boardH,pieceW,pieceH, edgeMap[visualIndex], pieceId);
      const wrap = document.createElement('div'); wrap.style.width = Math.max(48, Math.min(120, pieceW*0.9)) + 'px'; wrap.style.height = Math.max(36, Math.min(90, pieceH*0.9)) + 'px'; wrap.style.display='flex'; wrap.style.alignItems='center'; wrap.style.justifyContent='center'; wrap.style.borderRadius='8px'; wrap.style.background='linear-gradient(180deg,#fff,#fff)'; wrap.style.padding='6px'; wrap.style.boxShadow='0 6px 18px rgba(11,11,11,0.04)'; wrap.appendChild(svg);
      svg.dataset.origIndex = visualIndex; svg.dataset.pieceId = pieceId; pieceElements[pieceId] = svg; svg.addEventListener('dragstart', onDragStart); svg.addEventListener('dragend', onDragEnd); pool.appendChild(wrap);
    }
  }

  function onDragStart(e){ const svg = e.currentTarget; try{ const xml = new XMLSerializer().serializeToString(svg); const svg64 = btoa(unescape(encodeURIComponent(xml))); const imgSrcData = 'data:image/svg+xml;base64,' + svg64; const dragImg = new Image(); dragImg.src = imgSrcData; document.body.appendChild(dragImg); e.dataTransfer.setDragImage(dragImg, parseInt(svg.getAttribute('width')||svg.style.width||50)/2, parseInt(svg.getAttribute('height')||svg.style.height||50)/2); setTimeout(()=> document.body.removeChild(dragImg), 0); }catch(err){} svg.classList.add('dragging'); }
  function onDragEnd(e){ const svg = e.currentTarget; svg.classList.remove('dragging'); }

  function onDropSlot(e){ e.preventDefault(); const pieceId = e.dataTransfer.getData('text/plain'); if(!pieceId) return; const slot = e.currentTarget; const slotIdx = parseInt(slot.dataset.index,10); if(placedMap[slotIdx]){ const oldPieceId = placedMap[slotIdx]; const oldSvg = pieceElements[oldPieceId]; if(oldSvg){ pool.appendChild(oldSvg.parentElement); oldSvg.style.position=''; oldSvg.style.left=''; oldSvg.style.top=''; } delete placedMap[slotIdx]; }
    const svg = pieceElements[pieceId]; if(!svg) return; const left = parseFloat(slot.dataset.left), top = parseFloat(slot.dataset.top); svg.style.position='absolute'; svg.style.left = left + 'px'; svg.style.top = top + 'px'; svg.style.width = (getBoardGeometry().pieceW) + 'px'; svg.style.height = (getBoardGeometry().pieceH) + 'px'; board.appendChild(svg); placedMap[slotIdx] = pieceId; svg.setAttribute('draggable','false');
    // if piece is correct for this slot, animate snap (silent bounce)
    const orig = parseInt(svg.dataset.origIndex,10);
    if(orig === slotIdx){ svg.classList.add('snap'); setTimeout(()=> svg.classList.remove('snap'), 420); }
    checkSolved(); }

  function checkSolved(){ for(let i=0;i<TOTAL;i++){ if(!placedMap[i]) return false; const pid = placedMap[i]; const svg = pieceElements[pid]; if(!svg) return false; const orig = parseInt(svg.dataset.origIndex,10); if(orig !== i) return false; } clearInterval(timerId); show(`Solved on attempt ${attempt}! ${timeLeft}s remaining.`); hintBtn.disabled = true; retryBtn.disabled = true; startBtn.disabled = false; Object.values(pieceElements).forEach(s=> s.setAttribute('draggable','false')); return true; }

  function clearPlaced(){ Object.values(pieceElements).forEach(svg=>{ if(svg.parentElement === board){ const wrap = document.createElement('div'); wrap.style.width = Math.max(48, Math.min(120, getBoardGeometry().pieceW*0.9)) + 'px'; wrap.style.height = Math.max(36, Math.min(90, getBoardGeometry().pieceH*0.9)) + 'px'; wrap.style.display = 'flex'; wrap.style.alignItems='center'; wrap.style.justifyContent='center'; wrap.style.padding='6px'; wrap.style.borderRadius='8px'; wrap.appendChild(svg); pool.appendChild(wrap); svg.style.position=''; svg.style.left=''; svg.style.top=''; svg.style.width=''; svg.style.height=''; svg.setAttribute('draggable','true'); } }); placedMap = {}; }

  function startTimer(){ clearInterval(timerId); timeLeft = ATTEMPT_TIME; updateBadges(); timerId = setInterval(()=>{ timeLeft--; updateBadges(); if(timeLeft<=0){ clearInterval(timerId); onTimeout(); } },1000); }
  function onTimeout(){ show('Time up for this attempt.'); attempt++; updateBadges(); if(attempt < MAX_ATTEMPTS){ show(`Attempt ${attempt} ended. Preparing attempt ${attempt+1}...`); setTimeout(()=> prepareNewAttempt(), 900); } else { lockGame(); } }
  function prepareNewAttempt(){ clearPlaced(); createMapping(); buildBoardAndPieces(); startTimer(); hintBtn.disabled = attempt >= MAX_ATTEMPTS; retryBtn.disabled = attempt >= MAX_ATTEMPTS; show(`Attempt ${attempt+1} started.`); }
  function startGame(){ if(!imgSrc){ alert('Load an image first (file or URL).'); return; } generateEdgeMap(); attempt = 1; updateBadges(); createMapping(); buildBoardAndPieces(); startTimer(); hintBtn.disabled = false; retryBtn.disabled = false; show('Game started. Good luck!'); startBtn.disabled = true; }
  function retryAttempt(){ if(attempt >= MAX_ATTEMPTS) return; clearInterval(timerId); clearPlaced(); createMapping(); buildBoardAndPieces(); timeLeft = ATTEMPT_TIME; updateBadges(); startTimer(); show(`Attempt ${attempt} restarted.`); }
  function showHint(){ if(attempt >= MAX_ATTEMPTS) return; const overlay = document.createElement('div'); overlay.className = 'overlayHint'; const img = document.createElement('img'); img.className='fullImage'; img.src = imgSrc; overlay.appendChild(img); document.body.appendChild(overlay); setTimeout(()=> overlay.remove(),2000); }
  function lockGame(){ clearInterval(timerId); show('No attempts left — game locked.'); hintBtn.disabled = true; retryBtn.disabled = true; startBtn.disabled = false; Object.values(pieceElements).forEach(s=> s.setAttribute('draggable','false')); }

  loadBtn.addEventListener('click', ()=>{ if(imgFile.files && imgFile.files[0]){ const r=new FileReader(); r.onload = e => { imgSrc = e.target.result; show('Loaded image file. Press Start.'); }; r.readAsDataURL(imgFile.files[0]); } else if(imgUrl.value.trim()){ imgSrc = imgUrl.value.trim(); const t=new Image(); t.onload = ()=> show('Loaded image URL. Press Start.'); t.onerror = ()=> show('Failed to load image URL.'); t.src = imgSrc; } else { alert('Choose a file or enter a URL.'); } attempt = 0; timeLeft = ATTEMPT_TIME; updateBadges(); hintBtn.disabled = true; retryBtn.disabled = true; startBtn.disabled = false; });

  startBtn.addEventListener('click', startGame);
  hintBtn.addEventListener('click', showHint);
  retryBtn.addEventListener('click', retryAttempt);

  // resize handling: rebuild pieces and re-place previously placed items
  let resizeTimer = null;
  window.addEventListener('resize', ()=>{ clearTimeout(resizeTimer); resizeTimer = setTimeout(()=>{ if(!imgSrc) return; const prevPlaced = {...placedMap}; if(attempt>0 && attempt <= MAX_ATTEMPTS){ buildBoardAndPieces(); for(const slotIdxStr in prevPlaced){ const slotIdx = parseInt(slotIdxStr,10); const prevPieceId = prevPlaced[slotIdx]; const svg = pieceElements[prevPieceId]; if(svg){ const slot = board.querySelector(`.slot[data-index='${slotIdx}']`); if(slot){ const left = parseFloat(slot.dataset.left), top = parseFloat(slot.dataset.top); svg.style.position='absolute'; svg.style.left = left + 'px'; svg.style.top = top + 'px'; svg.style.width = (getBoardGeometry().pieceW) + 'px'; svg.style.height = (getBoardGeometry().pieceH) + 'px'; board.appendChild(svg); placedMap[slotIdx] = prevPieceId; svg.setAttribute('draggable','false'); } } } } else { buildBoardAndPieces(); } },150); });

  // init
  applyVariant(variantSel.value); updateBadges(); hintBtn.disabled = true; retryBtn.disabled = true; show('Load an image and press Start.');

})();
</script>
